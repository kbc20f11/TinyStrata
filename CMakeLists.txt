cmake_minimum_required(VERSION 3.24)
project(TinyStrata VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(TINYSTRATA_BACKEND CUDA CACHE STRING "Backend: CUDA or HIP (NONE configures without native inference)")
set_property(CACHE TINYSTRATA_BACKEND PROPERTY STRINGS CUDA HIP NONE)
option(STRATA_NATIVE_EXPERTS "Build the CPU expert pool for hybrid inference" ON)
option(STRATA_PREFILL_MMQ "Build GGML MMQ prompt kernels" ON)
option(STRATA_MMQ_KQUANTS "Build K-quant MMQ prompt kernels" ON)
option(STRATA_Q6K_EXPERTS "Build Q6_K expert kernels" ON)
option(TINYSTRATA_VISION "Build the image encoder for Qwen3.6" ON)
option(TINYSTRATA_GEMMA4 "Build the Gemma 4 26B-A4B image/MTP engine (CUDA/HIP)" ON)
option(TINYSTRATA_LFM2 "Build the LFM2.5-24B-A2B text engine (CUDA/HIP)" ON)
option(STRATA_PORTABLE "Build for other PCs with an AVX2 floor" OFF)
set(STRATA_ISA_FLOOR "" CACHE STRING "Optional older CPU floor: avx or none")
if(NOT STRATA_ISA_FLOOR MATCHES "^(|avx|none)$")
  message(FATAL_ERROR "STRATA_ISA_FLOOR must be empty, avx, or none")
endif()
if(STRATA_PORTABLE AND STRATA_ISA_FLOOR)
  message(FATAL_ERROR "STRATA_PORTABLE and STRATA_ISA_FLOOR exclude each other")
endif()
if(STRATA_PORTABLE AND MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
add_library(strata_warnings INTERFACE)
target_compile_options(strata_warnings INTERFACE
  $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CXX_COMPILER_ID:MSVC>>:/W3>
  $<$<AND:$<COMPILE_LANGUAGE:CXX>,$<NOT:$<CXX_COMPILER_ID:MSVC>>>:-Wall;-Wextra>)
add_library(strata_artifact INTERFACE)
target_include_directories(strata_artifact INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(strata_artifact INTERFACE strata_warnings)
if(TINYSTRATA_BACKEND STREQUAL "NONE")
  return()
elseif(TINYSTRATA_BACKEND STREQUAL "CUDA")
  set(STRATA_ENABLE_CUDA ON)
  set(_tinystrata_default_cuda_archs OFF)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    # The portable default must contain an sm_89+ image for native FP8 auto
    # selection on Ada/Blackwell. Finalize the list once the toolkit is known.
    set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "CUDA architectures")
    set(_tinystrata_default_cuda_archs ON)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(_tinystrata_default_cuda_archs)
    set(_tinystrata_cuda_archs 86)
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.8)
      list(APPEND _tinystrata_cuda_archs 89)
    endif()
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 12.8)
      list(APPEND _tinystrata_cuda_archs 120)
    endif()
    set(CMAKE_CUDA_ARCHITECTURES "${_tinystrata_cuda_archs}" CACHE STRING "CUDA architectures" FORCE)
  endif()
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(_strata_gpu_runtime_target CUDA::cudart)
  set(_strata_gpu_blas_target CUDA::cublas)
elseif(TINYSTRATA_BACKEND STREQUAL "HIP")
  set(STRATA_ENABLE_HIP ON)
  include(cmake/hip_backend.cmake)
  set(CMAKE_HIP_STANDARD 20)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  set(_strata_gpu_runtime_target strata_hip_runtime)
  set(_strata_gpu_blas_target ${STRATA_HIP_BLAS_TARGETS})
else()
  message(FATAL_ERROR "TINYSTRATA_BACKEND must be CUDA, HIP, or NONE")
endif()
if(STRATA_Q6K_EXPERTS)
  add_compile_definitions(STRATA_Q6K_EXPERTS=1)
endif()
if(STRATA_PREFILL_MMQ AND NOT STRATA_NATIVE_EXPERTS)
  message(FATAL_ERROR "MMQ requires STRATA_NATIVE_EXPERTS=ON for the pinned GGML source")
endif()
include(cmake/cpu.cmake)
if(TINYSTRATA_VISION)
  include(cmake/vision.cmake)
endif()

add_library(strata_kernels STATIC
  src/kernels/cuda/native_mmvq.cu src/kernels/cuda/iq_kernels.cu
  src/kernels/cuda/native_router.cu
  src/kernels/cuda/dequant_bf16.cu)
target_include_directories(strata_kernels PUBLIC include PRIVATE third_party/ggml)
target_link_libraries(strata_kernels PUBLIC strata_artifact ${_strata_gpu_runtime_target})
add_library(strata_prefill STATIC src/prefill/gemm.cu)
target_include_directories(strata_prefill PUBLIC include)
target_link_libraries(strata_prefill PUBLIC strata_kernels ${_strata_gpu_blas_target})
if(STRATA_ENABLE_HIP)
  target_sources(strata_prefill PRIVATE src/prefill/wmma_gemm.cu)
  if(STRATA_HIPBLASLT_AVAILABLE)
    target_compile_definitions(strata_prefill PRIVATE STRATA_HIPBLASLT_AVAILABLE=1)
    target_link_libraries(strata_prefill PRIVATE roc::hipblaslt)
  endif()
endif()
if(STRATA_PREFILL_MMQ)
  include(cmake/mmq.cmake)
else()
  target_sources(strata_prefill PRIVATE src/prefill/mmq_stub.cpp)
endif()
add_library(tinystrata_engine STATIC src/qwen35/q36_engine.cu)
if(STRATA_ENABLE_CUDA)
  # Preserve precise attention arithmetic: no -use_fast_math on this library.
  add_library(tinystrata_attention STATIC src/qwen35/int8_attention.cu src/qwen35/fp8_attention.cu)
  target_include_directories(tinystrata_attention PUBLIC include)
  target_link_libraries(tinystrata_attention PUBLIC CUDA::cudart)
  target_link_libraries(tinystrata_engine PRIVATE tinystrata_attention)
  add_library(tinystrata_fp8_attention STATIC src/common/fp8_attention.cu)
  target_include_directories(tinystrata_fp8_attention PUBLIC include)
  target_link_libraries(tinystrata_fp8_attention PUBLIC CUDA::cudart)
elseif(STRATA_ENABLE_HIP)
  add_library(tinystrata_attention STATIC src/qwen35/int8_attention_hip.cu src/qwen35/fp8_attention_hip.cu)
  target_include_directories(tinystrata_attention PUBLIC include)
  target_link_libraries(tinystrata_attention PUBLIC ${_strata_gpu_runtime_target})
  target_link_libraries(tinystrata_engine PRIVATE tinystrata_attention)
  add_library(tinystrata_fp8_attention STATIC src/common/fp8_attention_hip.cu)
  target_include_directories(tinystrata_fp8_attention PUBLIC include)
  target_link_libraries(tinystrata_fp8_attention PUBLIC ${_strata_gpu_runtime_target})
endif()
target_include_directories(tinystrata_engine PUBLIC include)
target_link_libraries(tinystrata_engine PRIVATE strata_prefill strata_warnings)
if(STRATA_NATIVE_EXPERTS)
  target_link_libraries(tinystrata_engine PRIVATE strata_kernels_cpu)
  target_compile_definitions(tinystrata_engine PRIVATE STRATA_Q36_CPU_POOL=1)
endif()
add_executable(tinystrata src/qwen35/q36_main.cpp)
target_link_libraries(tinystrata PRIVATE tinystrata_engine ${_strata_gpu_runtime_target})
if(STRATA_ENABLE_CUDA)
  foreach(target strata_kernels strata_prefill tinystrata_engine)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;-use_fast_math>)
  endforeach()
endif()
install(TARGETS tinystrata RUNTIME DESTINATION bin)
if(TINYSTRATA_LFM2 AND (STRATA_ENABLE_CUDA OR STRATA_ENABLE_HIP))
  if(NOT STRATA_NATIVE_EXPERTS)
    message(FATAL_ERROR "LFM2 hybrid inference requires STRATA_NATIVE_EXPERTS=ON")
  endif()
  set(_lfm_gg "${STRATA_LLAMA_SOURCE_DIR}/ggml")
  set(_lfm_cuda "${_lfm_gg}/src/ggml-cuda")
  set(_lfm_mmq_sources src/lfm2/mmq.cu src/prefill/ggml_cuda_host.cu
    "${_lfm_cuda}/quantize.cu")
  foreach(_type q4_0 q4_k q6_k q8_0)
    list(APPEND _lfm_mmq_sources "${_lfm_cuda}/template-instances/mmq-instance-${_type}.cu")
  endforeach()
  if(STRATA_ENABLE_HIP)
    # Same pinned GGML HIP MMQ configuration as Strata/MoMoE's prompt path.
    set_source_files_properties(${_lfm_mmq_sources} PROPERTIES LANGUAGE HIP)
  endif()
  add_library(tinystrata_lfm2_mmq STATIC ${_lfm_mmq_sources})
  target_include_directories(tinystrata_lfm2_mmq PRIVATE
    "${_lfm_gg}/include" "${_lfm_gg}/src" "${_lfm_cuda}" include)
  target_compile_options(tinystrata_lfm2_mmq PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)
  if(MSVC)
    target_compile_options(tinystrata_lfm2_mmq PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/bigobj>)
  endif()
  if(STRATA_ENABLE_HIP)
    target_compile_definitions(tinystrata_lfm2_mmq PRIVATE GGML_USE_HIP=1)
    # GGML already provides the HIP compatibility macros used by its MMQ code.
    target_link_libraries(tinystrata_lfm2_mmq PUBLIC ggml-base hip::host ${_strata_gpu_blas_target})
  else()
    target_link_libraries(tinystrata_lfm2_mmq PUBLIC ggml-base CUDA::cudart)
  endif()
  add_library(tinystrata_lfm2 STATIC src/lfm2/engine.cpp src/lfm2/lfm2moe.cpp
    src/lfm2/cpu_experts.cpp src/lfm2/kernels.cu src/lfm2/matvec.cu)
  # Match the quantized matvec implementation's rounding/reciprocal contract.
  # Attention and recurrent kernels retain their normal CUDA arithmetic.
  if(STRATA_ENABLE_CUDA)
    set_source_files_properties(src/lfm2/matvec.cu PROPERTIES COMPILE_OPTIONS "--use_fast_math")
  endif()
  target_include_directories(tinystrata_lfm2 PRIVATE include third_party/ggml "${_lfm_gg}/include")
  # Keep default CUDA arithmetic for the new path until model comparisons pass.
  target_compile_options(tinystrata_lfm2 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)
  target_link_libraries(tinystrata_lfm2 PRIVATE strata_artifact strata_kernels tinystrata_lfm2_mmq
    strata_kernels_cpu ggml-cpu ggml-base ${_strata_gpu_runtime_target} ${_strata_gpu_blas_target} Threads::Threads)
  if(STRATA_ENABLE_CUDA)
    target_link_libraries(tinystrata_lfm2 PRIVATE CUDA::cuda_driver)
  endif()
  target_link_libraries(tinystrata_lfm2 PRIVATE tinystrata_fp8_attention)
  add_executable(tinystrata-lfm2 src/lfm2/main.cpp)
  target_include_directories(tinystrata-lfm2 PRIVATE include)
  target_link_libraries(tinystrata-lfm2 PRIVATE tinystrata_lfm2 ${_strata_gpu_runtime_target})
  install(TARGETS tinystrata-lfm2 RUNTIME DESTINATION bin)
endif()
if(TINYSTRATA_GEMMA4 AND (STRATA_ENABLE_CUDA OR STRATA_ENABLE_HIP))
  include(cmake/gemma4.cmake)
endif()
if(TARGET strata_kq)
  install(TARGETS strata_kq RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
endif()
install(FILES LICENSE NOTICE DESTINATION share/licenses/TinyStrata)
install(FILES third_party/ggml/LICENSE DESTINATION share/licenses/TinyStrata/ggml)
install(FILES third_party/licenses/FlashInfer-BSD-3-Clause.txt DESTINATION share/licenses/TinyStrata/FlashInfer)
